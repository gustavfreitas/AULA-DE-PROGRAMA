//Escreva uma função que calcula o número fatorial de um número. Implementar uma versão recursiva.

int	recursivo(int n){

	if (n <= 0) {
		return 1;
	}
	else {
		return n * recursivo(n - 1);
	}
}
int main() {
	int resultado = recursivo(5);
	printf("%d", resultado);
	return 0;
}

// E uma versão com laço de repetição.

int main(){
	int n;
	for (n = 0; n < 5; n++) {
		printf("%d\n", n);
	}

	return 0;
}
