#include <stdio.h>

// 2 - Fazer um programa em C que imprime uma tabela com a tabuada de 1 a 9.

int main() {
    int numero;

    printf("Digite um número para gerar a tabuada: ");
    scanf_s("%d", &numero);

    printf("Tabuada do %d:\n", numero);
    for (int i = 1; i <= 10; i++) {
        printf("%d x %d = %d\n", numero, i, numero * i);
    }

    return 0;
}
