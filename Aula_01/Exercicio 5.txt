#include <stdio.h>

//5 - Usar uma variável double para ler o valor em Fahrenheit e a fórmula
//C = (f - 32.0) * (5.0 / 9.0)

int main() {
    double Fahrenheit;
    double C;

    printf("Digite a temperatura em Fahrenheit: ");
    scanf_s("%lf", &Fahrenheit);

    C = (Fahrenheit - 32.0) * (5.0 / 9.0);
    printf("A Temperatura em Celsius e: %lf", C);
  

    return 0;
}
